# wheels exist for any torch release). See requirements.txt / AGENTS.md for
# the Apple Silicon (arm64) pins, which do not have these constraints.
#

#   conda create -n viz -c conda-forge python=3.11 -y
#   conda activate viz
#   pip install jupyter ipykernel
#   pip install -r requirements-intel-mac.txt
